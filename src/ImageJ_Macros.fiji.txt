// Macros for CODEX Machine Learning-based Segmentation Processing

// These macros are designed to aid in the post-processing of CODEX machine learning-based segmentation output.
// To load them in to ImageJ, select Startup Macros and insert the below text into the document.


macro "Segmentation Post-processing loop [1]" {
imageCount = getNumber("How many images are you processing?", 0);
for (i = 0; i < imageCount; i++) {
run("Threshold...");
setThreshold(0, 0);
run("Convert to Mask");
run("Invert");
run("Watershed");
run("Save");
run("Open Next");
}
}


macro "Marker Intensity (Color Images) [2]"{
imageCount = getNumber("How many images are you processing?", 0);
for (i = 0; i < imageCount; i++) {
roiManager("Deselect");
roiManager("Measure");
run("Open Next");
}
}


macro "Greyscale Area Percentage (Array) [3]"{
imageCount = getNumber("How many images are you processing?", 0);
threshold = newArray(50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50, 50);
for (i = 0; i < imageCount; i++) {
if (threshold[i] == 0) {
setAutoThreshold("Default");
run("Threshold...");
setOption("BlackBackground", true);
run("Convert to Mask");
}
else {
setAutoThreshold("Default");
run("Threshold...");
setOption("BlackBackground", true);
setThreshold(0, threshold[i]);
run("Convert to Mask");
}

run("Invert");
run("Save");
roiManager("Deselect");
roiManager("Measure");
run("Open Next");
}
}
